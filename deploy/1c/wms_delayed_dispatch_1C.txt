// Обработчик регламентного задания: МВ_ДоработкиСервер.ОтправитьОтложенныеЗаказыПеремещенияВWMS
// Расписание: каждые 1-5 минут; запретить повторный запуск.

Процедура ОтправитьОтложенныеЗаказыПеремещенияВWMS() Экспорт

	ГраницаОтправки = ТекущаяДата() - 20 * 60;

	Запрос = Новый Запрос;
	Запрос.Текст =
	"ВЫБРАТЬ ПЕРВЫЕ 100
	| Заказы.Ссылка КАК Ссылка,
	| Заказы.Дата КАК Дата,
	| Заказы.Проведен КАК Проведен
	|ИЗ
	| Документ.ЗаказНаПеремещение КАК Заказы
	|  ЛЕВОЕ СОЕДИНЕНИЕ РегистрСведений.WMSСтатусыДокументов.СрезПоследних КАК СтатусыWMS
	|  ПО Заказы.Ссылка = СтатусыWMS.Объект
	|ГДЕ
	| НЕ Заказы.ПометкаУдаления
	| И Заказы.Проведен
	| И Заказы.Дата <= &ГраницаОтправки
	| И ПОДСТРОКА(Заказы.Комментарий, 1, 6) = ""[MOVA]""
	| И ЕСТЬNULL(СтатусыWMS.ОписаниеСтатуса, """") = """"
	|УПОРЯДОЧИТЬ ПО
	| Заказы.Дата";
	Запрос.УстановитьПараметр("ГраницаОтправки", ГраницаОтправки);

	Выборка = Запрос.Выполнить().Выбрать();
	Пока Выборка.Следующий() Цикл
		НачатьТранзакцию();
		Попытка
			Блокировка = Новый БлокировкаДанных;
			ЭлементБлокировки = Блокировка.Добавить("Документ.ЗаказНаПеремещение");
			ЭлементБлокировки.УстановитьЗначение("Ссылка", Выборка.Ссылка);
			Блокировка.Заблокировать();

			// Пока ожидали блокировку, пользователь мог отменить заказ или WMS
			// уже могла принять его в другом сеансе.
			Заказ = Выборка.Ссылка;
			Если Заказ.ПометкаУдаления
				Или НЕ Заказ.Проведен
				Или НЕ ПустаяСтрока(ПолучитьСтатусWMSОтложенногоЗаказа(Заказ)) Тогда
				ЗафиксироватьТранзакцию();
				Продолжить;
			КонецЕсли;

			WMSОбменыИнформацией.ВыгрузитьДокументВWMS(
				Выборка.Ссылка,
				Выборка.Дата,
				Выборка.Проведен
			);

			// Выгрузка считается успешной только когда WMS-модуль записал статус.
			Если ПустаяСтрока(ПолучитьСтатусWMSОтложенногоЗаказа(Выборка.Ссылка)) Тогда
				ЗаписьЖурналаРегистрации(
					"MOVA.WMS.ОтложеннаяОтправка",
					УровеньЖурналаРегистрации.Предупреждение,
					,
					Выборка.Ссылка,
					"WMS не подтвердила отправку; заказ останется доступным и будет повторен следующим запуском"
				);
			КонецЕсли;
			ЗафиксироватьТранзакцию();
		Исключение
			ТекстОшибки = ОписаниеОшибки();
			ОтменитьТранзакцию();
			ЗаписьЖурналаРегистрации(
				"MOVA.WMS.ОтложеннаяОтправка",
				УровеньЖурналаРегистрации.Ошибка,
				,
				Выборка.Ссылка,
				ТекстОшибки
			);
		КонецПопытки;
	КонецЦикла;

КонецПроцедуры

Функция ПолучитьСтатусWMSОтложенногоЗаказа(ЗаказСсылка) Экспорт

	Запрос = Новый Запрос;
	Запрос.Текст =
	"ВЫБРАТЬ ПЕРВЫЕ 1
	| СтатусыWMS.ОписаниеСтатуса КАК СтатусWMS
	|ИЗ
	| РегистрСведений.WMSСтатусыДокументов.СрезПоследних КАК СтатусыWMS
	|ГДЕ
	| СтатусыWMS.Объект = &Заказ";
	Запрос.УстановитьПараметр("Заказ", ЗаказСсылка);

	Выборка = Запрос.Выполнить().Выбрать();
	Если Выборка.Следующий() И ЗначениеЗаполнено(Выборка.СтатусWMS) Тогда
		Возврат СокрЛП(Строка(Выборка.СтатусWMS));
	КонецЕсли;

	Возврат "";

КонецФункции
// Подписка на событие ПередЗаписью документа ЗаказНаПеремещение.
// Источник подписки: ДокументОбъект.ЗаказНаПеремещение.
Процедура ЗапретитьИзменениеПереданныхВWMSЗаказовПеремещения(Источник, Отказ, РежимЗаписи, РежимПроведения) Экспорт

	Если Лев(СокрЛП(Строка(Источник.Комментарий)), 6) <> "[MOVA]" Тогда
		Возврат;
	КонецЕсли;
	Если НЕ ЗначениеЗаполнено(Источник.Ссылка) Тогда
		Возврат;
	КонецЕсли;

	СтатусWMS = ПолучитьСтатусWMSОтложенногоЗаказа(Источник.Ссылка);
	Если НЕ ПустаяСтрока(СтатусWMS) Тогда
		Отказ = Истина;
		ВызватьИсключение "Замовлення вже передано до WMS і заблоковано. Поточний статус: " + СтатусWMS;
	КонецЕсли;
